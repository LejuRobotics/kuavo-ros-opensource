cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(wheel_arm_abs_vel LANGUAGES CXX)
  set(DIRECT_TCP_STANDALONE ON)
  enable_testing()
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(pinocchio REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp roslib rosbag std_msgs geometry_msgs std_srvs kuavo_msgs noitom_hi5_hand_udp_python message_generation)
set(DIRECT_TCP_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(DIRECT_TCP_MESSAGE_INCLUDE "" CACHE PATH "Matching generated wheel_arm_abs_vel messages for standalone builds")
if(DIRECT_TCP_STANDALONE)
  add_message_files(DIRECTORY ${DIRECT_TCP_ROOT}/msg FILES DirectControllerInput.msg TeleopTcpTarget.msg)
  generate_messages(DEPENDENCIES geometry_msgs std_msgs)
  include_directories(${CATKIN_DEVEL_PREFIX}/include)
endif()
add_library(absolute_direct_controller_tcp
  ${DIRECT_TCP_ROOT}/src/direct_tcp_mapper.cpp
  ${DIRECT_TCP_ROOT}/src/direct_tcp_model.cpp)
target_include_directories(absolute_direct_controller_tcp PUBLIC ${DIRECT_TCP_ROOT}/include ${catkin_INCLUDE_DIRS})
target_link_libraries(absolute_direct_controller_tcp PUBLIC Eigen3::Eigen pinocchio::pinocchio OpenSSL::Crypto nlohmann_json::nlohmann_json ${roslib_LIBRARIES})
target_compile_options(absolute_direct_controller_tcp PRIVATE -Wall -Wextra -Werror=return-type)

if(DIRECT_TCP_STANDALONE OR CATKIN_ENABLE_TESTING)
  find_package(GTest REQUIRED)
  add_executable(direct_tcp_gtest ${DIRECT_TCP_ROOT}/test/direct_tcp_gtest.cpp)
  target_link_libraries(direct_tcp_gtest PRIVATE absolute_direct_controller_tcp GTest::GTest GTest::Main)
  add_test(NAME direct_tcp_gtest COMMAND direct_tcp_gtest)
  add_executable(direct_tcp_json_sample_test ${DIRECT_TCP_ROOT}/test/direct_tcp_json_sample_test.cpp)
  target_link_libraries(direct_tcp_json_sample_test PRIVATE absolute_direct_controller_tcp GTest::GTest GTest::Main)
  add_test(NAME direct_tcp_json_sample_test COMMAND direct_tcp_json_sample_test)
  # Bind the explicit checked-in test fixture to this checkout, independently
  # of the container mount path. Production configuration has no default path.
  file(READ ${DIRECT_TCP_ROOT}/config/direct_tcp/400063_simulation.json DIRECT_TCP_TEST_JSON)
  string(REPLACE "/current/src/kuavo_assets" "${DIRECT_TCP_ROOT}/../kuavo_assets"
    DIRECT_TCP_TEST_JSON "${DIRECT_TCP_TEST_JSON}")
  string(REPLACE "/current/src/manipulation_nodes/motion_capture_ik" "${DIRECT_TCP_ROOT}" DIRECT_TCP_TEST_JSON "${DIRECT_TCP_TEST_JSON}")
  string(REPLACE "models/biped_s400063_v7_frozen.urdf" "${DIRECT_TCP_ROOT}/config/direct_tcp/models/biped_s400063_v7_frozen.urdf" DIRECT_TCP_TEST_JSON "${DIRECT_TCP_TEST_JSON}")
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/direct_tcp_test_config.json "${DIRECT_TCP_TEST_JSON}")
  set_tests_properties(direct_tcp_gtest PROPERTIES ENVIRONMENT
    "DIRECT_TCP_CONFIG=${CMAKE_CURRENT_BINARY_DIR}/direct_tcp_test_config.json")
endif()

add_executable(direct_controller_tcp_node ${DIRECT_TCP_ROOT}/src/direct_controller_tcp_node.cpp)
